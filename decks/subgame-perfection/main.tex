% One instruction. The label carries the accent colour; its column has to be
% wide enough for a word, so the width is settable.
\newcommand{\step}[3][1.2cm]{%
  \noindent\makebox[#1][l]{\color{inkblue}\Large\bfseries #2}%
  \begin{minipage}[t]{\dimexpr\linewidth-#1\relax}\raggedright\large #3%
  \end{minipage}\par\vspace{0.55em}}

% The centipede tree. Nodes are named by the player who moves and a letter, so
% that a strategy can be written as one action per named node. The argument is
% drawn afterwards, which lets a page add its own marks.
\newcommand{\centipede}[2][1.12]{%
  \begin{tikzpicture}[font=\small, scale=#1,
      every node/.style={transform shape},
      dec/.style={circle, draw=inkblue, line width=1.1pt, fill=white,
                  text=inkblue, minimum size=1.35cm, font=\bfseries\small,
                  inner sep=0pt},
      leaf/.style={font=\normalsize},
      writebox/.style={draw=inkfaint, line width=0.7pt, rounded corners=2pt,
                   fill=white}]
    \foreach \who/\lab [count=\i from 0] in {1/A, 2/B, 1/C, 2/D}{%
      \node[dec] (n\i) at (\i*4.6,0) {\who:\lab};}
    \node[leaf] (end) at (18.4,0) {\((4,4)\)};
    \foreach \pay [count=\i from 0] in {{\((2,0)\)}, {\((1,3)\)}, {\((4,2)\)},
                                        {\((3,5)\)}}{%
      \draw[-{Stealth[length=3mm]}, inkred, line width=1.2pt]
        (n\i) -- node[left, font=\small, text=inkred] {take} ++(0,-2.1);
      \node[leaf] at (\i*4.6,-2.65) {\pay};}
    \foreach \i in {0,1,2}{%
      \pgfmathtruncatemacro{\j}{\i+1}
      \draw[-{Stealth[length=3mm]}, inkteal, line width=1.2pt]
        (n\i) -- node[above, font=\small, text=inkteal] {pass} (n\j);}
    \draw[-{Stealth[length=3mm]}, inkteal, line width=1.2pt]
      (n3) -- node[above, font=\small, text=inkteal] {pass} (end);
    #2
  \end{tikzpicture}}

% A box above each node, for writing one action per node onto the tree.
\newcommand{\nodeboxes}{%
  \foreach \i in {0,...,3}{%
    \draw[writebox] (\i*4.6-1.1,1.1) rectangle (\i*4.6+1.1,1.9);}}

% A strategy, as the class writes it on paper: one row per node.
\newcommand{\strategytable}[3]{%
  \begin{tabular}{@{}c@{\hspace{0.9cm}}c@{}}
    \multicolumn{2}{@{}c@{}}{\large\bfseries\color{inkblue} #1}\\[0.55em]
    {\small\color{inkgrey} Node} & {\small\color{inkgrey} Action}\\[0.3em]

    {\large #3} & \blank[3.2]\\
  \end{tabular}}

\newcommand{\strategylist}[3]{%

    & {\small\color{inkgrey} #2} & {\small\color{inkgrey} #3}\\[0.25em]
    {\large\color{inkblue} 1} & \blank[2.4] & \blank[2.4]\\[0.45em]
    {\large\color{inkblue} 2} & \blank[2.4] & \blank[2.4]\\[0.45em]
    {\large\color{inkblue} 3} & \blank[2.4] & \blank[2.4]\\[0.45em]
    {\large\color{inkblue} 4} & \blank[2.4] & \blank[2.4]\\
  \end{tabular}}

% One player's four by four payoff matrix. The label cells are blank too, since
% the class fills in the strategies it has just built. The scale is settable so
% that a page carrying instructions as well can still hold both matrices.
\newcommand{\payoffmatrix}[2][1]{%
  \begin{tikzpicture}[font=\small, scale=#1,
      every node/.style={transform shape},
      cell/.style={draw=inkfaint, line width=0.7pt, rounded corners=2pt,
                   fill=white}]
    \foreach \j in {0,...,3}{%
      \draw[cell] (\j*2.1,0.2) rectangle (\j*2.1+1.9,0.9);
      \foreach \i in {0,...,3}{%
        \draw[cell] (\j*2.1,-\i*1.15) rectangle (\j*2.1+1.9,-\i*1.15-0.95);}}
    \foreach \i in {0,...,3}{%
      \draw[cell] (-2.1,-\i*1.15) rectangle (-0.2,-\i*1.15-0.95);}
    \node[text=inkblue, font=\large\bfseries] at (3.0,1.6) {#2};
  \end{tikzpicture}}

\slidehead{The rules}

\vspace{0.5cm}

{\Large Each round has two phases.}

\vspace{0.6em}

\step[2.4cm]{Night}{The Traitors agree silently on one Faithful to remove. Everyone
  else has their eyes shut.}
\step[2.4cm]{Day}{Everyone votes. The most-voted player is banished and their role is
  revealed. Ties are broken at random.}

\vspace{0.5cm}

{\Large So every round costs two players: one banished, one murdered.}

\vspace{0.7cm}

\begin{center}

\slidehead{The roles}

\vspace{0.9cm}

\step{1}{Everyone is dealt a role, face down. Look at it. Do not show it to
  anybody.}
\step{2}{The Traitors learn who the other Traitors are. The Faithful learn
  nothing.}
\step{3}{Sit in a circle and stay in that order for the whole game. Your
  position matters later.}

\vspace{1.2cm}

\begin{center}
{\Large Today: \blank[2.2] players, of whom \blank[2.2] are Traitors.}
\end{center}

\vspace{1.2cm}

{\large\color{inkgrey} Everything a Faithful knows comes from how people vote.
That is the whole game.}

\newpage
\slidehead{Game 1: no guidance}

\vspace{1.1cm}

\step{1}{Play the two phases as described. Vote however you like.}
\step{2}{Nobody is telling you how to vote. Talk, argue, accuse.}
\step{3}{We stop after a round or two. The point is not to finish.}

\vspace{1.4cm}

{\Large\color{inkblue} The question to hold on to:}

\vspace{0.5em}

{\Large if the Traitors are working together, can you tell?}

\newpage
\slidehead{Game 2: Vote-Left}

\vspace{0.9cm}

{\Large New roles, fresh game, and now a rule everybody follows.}

\vspace{0.8em}

\step[3.8cm]{Rule}{Vote for the next surviving player to your left.}
\step[3.8cm]{Why}{Under the rule everyone receives exactly one vote, so the banishment
  is uniform: the same \(1/n\) chance as voting at random. It costs the
  Faithful nothing.}
\step[3.8cm]{Catch}{Break the rule and it shows. One player gets two votes and
  somebody gets none, and everybody can see who did it.}
\step[3.8cm]{Punishment}{Anyone seen to break the rule is banished next.}

\vspace{1cm}

{\large\color{inkgrey} Play this one on towards the end. Watch what happens as
the circle gets small.}

\newpage
\slidehead{Game 3: break it for fun}

\vspace{1.1cm}

{\Large Same rule, same punishment, with one change.}

\vspace{0.9em}

\step[1.6cm]{New}{Anyone may break Vote-Left now and then, for no reason at all. No
  warning, no announcement.}

\vspace{1.2cm}

{\Large\color{inkblue} The question to hold on to:}

\vspace{0.5em}

{\Large now that an odd vote might just be somebody playing around, is the
threat to banish a suspect still worth carrying out?}

\slidehead{Write down a strategy}

\vspace{0.4cm}

{\Large Pick a player. Write a strategy for them on paper, then swap and write
one for the other.}

\vspace{0.4em}

\step{1}{A strategy gives an action at \emph{every} node where that player
  moves, so the table has one row per node.}
\step{2}{Fill in both rows, even if the first action ends the game. A strategy
  says what you \emph{would} do, not only what happens.}

\newpage
\slidehead{Finding the Nash equilibria}

\vspace{0.4cm}

{\Large The pair \((A, B)\) is an ordinary normal form game, so we find its
equilibria the way we always do.}

\vspace{0.6em}

\step{1}{Down each column of \(A\), underline player 1's best payoff.}
\step{2}{Across each row of \(B\), underline player 2's best payoff.}
\step{3}{A position underlined in both matrices is a pure Nash equilibrium.
  There is more than one.}

\vspace{0.2cm}

\begin{center}
\payoffmatrix[0.78]{\(A\)} \hspace{2cm} \payoffmatrix[0.78]{\(B\)}
\end{center}

\newpage
\slidehead{Back to the tree}

\vspace{0.4cm}

{\Large Only one of those equilibria survives backward induction. The others
rest on a threat nobody would carry out.}

\vspace{0.6em}

\step{1}{Start at D, the last node, and mark the action player 2 prefers there.}
\step{2}{Work leftwards: at each node, compare the payoffs the marks to its
  right lead to.}
\step{3}{Read off one action per node. That pair of strategies is the subgame
  perfect equilibrium.}

\vspace{0.4cm}

\centipede{\nodeboxes}
\end{center}
