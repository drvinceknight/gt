% A block of write-on cells. Boxes rather than rules, because thin lines
% disappear on a projector. #1 is the list of column positions, #2 the cell
% width, #3 the number of rows.
\newcommand{\cellrows}[3]{%
  \foreach \row in {1,...,#3}{%
    \node[anchor=west, text=inktext, font=\bfseries] at (0,-0.95*\row) {\row};
    \foreach \col in {#1}{%
      \draw[inkfaint, line width=1pt, rounded corners=2.5pt, fill=white]
        (\col,-0.95*\row-0.33) rectangle (\col+#2,-0.95*\row+0.33);}}}

% Column heading, sitting above a thick accent rule.
\newcommand{\colhead}[1]{{\normalsize\bfseries\color{inktext}#1}}

% A one-line reminder of whichever game a page is about, so no page sends the
% students back to look for the matrices. #1 and #2 are the matrix bodies,
% #3 the trailing note.
\newcommand{\gameline}[3]{%
  {\small\color{inkgrey} The game:}\;
  {\large\(A = \left(\begin{smallmatrix} #1 \end{smallmatrix}\right)\)}\;
  {\large\(B = \left(\begin{smallmatrix} #2 \end{smallmatrix}\right)\)}\;
  {\small\color{inkgrey} #3}}

{\large Is \(c_1\) dominated by \(c_2\)? \blank[2] \hspace{2.2cm}
