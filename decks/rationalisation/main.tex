% A block of write-on cells. Boxes rather than rules, because thin lines
% disappear on a projector. #1 is the list of column positions, #2 the cell
% width, #3 the number of rows.
\newcommand{\cellrows}[3]{%
  \foreach \row in {1,...,#3}{%
    \node[anchor=west, text=inktext, font=\bfseries] at (0,-0.95*\row) {\row};
    \foreach \col in {#1}{%
      \draw[inkfaint, line width=1pt, rounded corners=2.5pt, fill=white]
        (\col,-0.95*\row-0.33) rectangle (\col+#2,-0.95*\row+0.33);}}}

% Column heading, sitting above a thick accent rule.
\newcommand{\colhead}[1]{{\normalsize\bfseries\color{inktext}#1}}

% A one-line reminder of whichever game a page is about, so no page sends the
% students back to look for the matrices. #1 and #2 are the matrix bodies,

  {\large\(B = \begin{pmatrix} #2 \end{pmatrix}\)}\;\;
  {\small\color{inkgrey} #3}}

\slidehead{The game}
\notesref{2.2.3}{Games: Definition: Normal Form Game}

\vspace{0.6cm}

\begin{center}
{\Large \(A = \begin{pmatrix} 2 & -2 \\ -1 & 1 \end{pmatrix}
 \qquad\qquad
 B = \begin{pmatrix} -2 & 2 \\ 1 & -1 \end{pmatrix}\)}
\end{center}

\vspace{1cm}

{\large \(A\) pays the row player, \(B\) pays the column player. Rows are
\(r_1\) and \(r_2\); columns are \(c_1\) and \(c_2\).}

\vspace{0.8cm}

{\large Every cell adds to \answerblank[1.5]{\(0\)}, so this is a
\answerrule[4cm]{zero sum} game.}

\vspace{0.8cm}

{\large The action sets are \(A_r =\) \answerblank[2.4]{\(\{r_1, r_2\}\)} and
\(A_c =\) \answerblank[2.4]{\(\{c_1, c_2\}\)}.}

\vspace{0.8cm}

{\large If the row player picks \(r_1\) and you pick \(c_2\), you score
\answerblank[1.5]{\(+2\)} and they score \answerblank[1.5]{\(-2\)}.}

{\large
\(x^{*} = \)\;\(\left\{\begin{array}{l}
  \text{\answerblank[1.4]{\(1\)} \; if } p < \text{\answerblank[1.4]{\(1/3\)}}\\[0.9em]

carrying into the algebra.}
