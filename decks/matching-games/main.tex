% One person's ranking, printed: the name, then the other side most preferred
% first.
\newcommand{\ranking}[5]{%
  \makebox[2.3cm][l]{\bfseries #1}#2 \(\succ\) #3 \(\succ\) #4 \(\succ\) #5}

% A numbered step, at whatever size the surrounding text is.
\newcommand{\gsstep}[2]{%
  \noindent\makebox[0.8cm][l]{\color{inkblue}\bfseries #1}%
  \begin{minipage}[t]{\dimexpr\linewidth-0.8cm\relax}\raggedright #2%
  \end{minipage}\par\vspace{0.4em}}

% The steps of the Gale-Shapley algorithm, for whichever side proposes.
\newcommand{\galeshapley}{%
  \gsstep{1}{Everyone on the proposing side who is not held proposes to the
    favourite they have not yet asked.}
  \gsstep{2}{Everyone proposed to holds the best offer they have, counting the
    one already held, and rejects the rest.}
  \gsstep{3}{Repeat until nobody is rejected. What is held is the matching.}}

% One person's preference list, waiting to be filled in. The four names after
% the person's own are the ranking the class is steered to, most preferred
% first, which the answer build writes in.
\newcommand{\prefrow}[5]{%
  \makebox[2.1cm][l]{#1}
  \answerblank[2]{#2}\, \answerblank[2]{#3}\, \answerblank[2]{#4}\,
  \answerblank[2]{#5}}

% A proposal in a round of deferred acceptance: one that is held, and one that
% is turned down. The arguments are the mathematician and the physicist.
\newcommand{\hold}[2]{%
  \draw[-{Stealth[length=2.2mm]}, inkteal, line width=1pt] (m#1) -- (p#2);}
\newcommand{\reject}[2]{%
  \draw[-{Stealth[length=2.2mm]}, inkred, line width=1pt, dashed]
    (m#1) -- (p#2);}

% A proposal from a physicist, when the physicists propose. The arguments are
% the physicist and the mathematician.
\newcommand{\phold}[2]{%
  \draw[-{Stealth[length=2.2mm]}, inkteal, line width=1pt] (p#1) -- (m#2);}
\newcommand{\preject}[2]{%
  \draw[-{Stealth[length=2.2mm]}, inkred, line width=1pt, dashed]
    (p#1) -- (m#2);}
