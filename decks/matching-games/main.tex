% The two sides of the market, with #1 drawn afterwards so each round can add
% its own proposals, holds and rejections.
\newcommand{\bipartite}[1]{%
  \begin{tikzpicture}[scale=1.5, every node/.style={transform shape},
      person/.style={draw=inkblue, line width=0.9pt, fill=white, text=inkblue,
                     rounded corners=3pt, minimum width=2.3cm,
                     minimum height=0.72cm, font=\small\bfseries}]
    \foreach \name [count=\i from 0] in {Gauss, Noether, Turing, Euler}{%
      \node[person] (m\i) at (0,-\i*1.25) {\name};}
    \foreach \name [count=\i from 0] in {Einstein, Curie, Newton, Feynman}{%
      \node[person, draw=inkteal, text=inkteal] (p\i) at (4.7,-\i*1.25)
        {\name};}
    #1
  \end{tikzpicture}}

% One person's preference list, waiting to be filled in. The four names after
% the person's own are the ranking the class is steered to, most preferred
% first, which the answer build writes in.
\newcommand{\prefrow}[5]{%
  \makebox[2.1cm][l]{#1}
  \answerblank[2]{#2}\, \answerblank[2]{#3}\, \answerblank[2]{#4}\,
  \answerblank[2]{#5}}

% A proposal in a round of deferred acceptance: one that is held, and one that
% is turned down. The arguments are the mathematician and the physicist.
\newcommand{\hold}[2]{%
  \draw[-{Stealth[length=2.2mm]}, inkteal, line width=1pt] (m#1) -- (p#2);}
\newcommand{\reject}[2]{%
  \draw[-{Stealth[length=2.2mm]}, inkred, line width=1pt, dashed]
    (m#1) -- (p#2);}

  both would walk away from what they have been given.}
