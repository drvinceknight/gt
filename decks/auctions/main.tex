% A number line for the argument about where the highest other bid b falls.
% #1 and #2 place your value v and your bid, on a line from 0 to 10; the answer
% build shades the stretch between them, with #3 as its label.
\newcommand{\bidline}[3]{%
  \begin{tikzpicture}[scale=0.95]
    \draw[inkgrey, line width=1pt, -{Stealth[length=2.5mm]}] (0,0) -- (10.6,0);
    \onlyanswer{%
      \fill[inkanswer, opacity=0.15] (#1,-0.3) rectangle (#2,0.3);
      \node[text=inkanswer, font=\small, align=center, text width=4.4cm]
        at ({(#1+#2)/2},-1.0) {#3};}
    \foreach \x/\lab in {#1/\(v\), #2/your bid}{%
      \draw[inkblue, line width=1.2pt] (\x,-0.3) -- (\x,0.3);
      \node[above, text=inkblue, font=\normalsize] at (\x,0.3) {\lab};}
  \end{tikzpicture}}

% Axes for one round's bids against values, both running from 0 to 12. The
% line bid = value is printed for reference; #2 is drawn on top, so the answer
% build can add the equilibrium bid.
\newcommand{\bidplot}[2]{%
  \begin{tikzpicture}[scale=0.54]
    \fill[white] (0,0) rectangle (12,12);
    \foreach \x in {1,...,12}{%
      \draw[inkfaint, line width=0.4pt] (\x,0) -- (\x,12);
      \draw[inkfaint, line width=0.4pt] (0,\x) -- (12,\x);}
    \draw[inkgrey, line width=1pt, -{Stealth[length=2.5mm]}] (0,0) -- (12.7,0)
      node[right, font=\small] {value};
    \draw[inkgrey, line width=1pt, -{Stealth[length=2.5mm]}] (0,0) -- (0,12.7)
      node[above, font=\small] {bid};
    \foreach \x in {0,3,6,9,12}{%
      \node[below, text=inkgrey, font=\small] at (\x,0) {\x};
      \node[left, text=inkgrey, font=\small] at (0,\x) {\x};}
    \draw[inkgrey, dashed, line width=0.8pt] (0,0) -- (12,12)
      node[above left, font=\small] {bid \(=\) value};
    \node[text=inkblue, font=\large\bfseries] at (6,13.4) {#1};
    #2
  \end{tikzpicture}}

\newpage
\slidehead{What are the rules, exactly?}
\notesref{16.2.1}{Auction Games: Definition: Auction Game}

\vspace{1cm}

{\large Everyone submits \answerrule[4cm]{one sealed bid}, without seeing
anyone else's, and the item goes to \answerrule[5cm]{the highest bidder}.}

\vspace{1.1cm}

{\large In a first-price auction the winner pays \answerrule[5cm]{their own bid}.
In a second-price auction they pay \answerrule[5cm]{the second highest bid}.}

\vspace{1.1cm}

{\large So in round 2, with value \(v\), your payoff is \(v -\)
\answerblank[4.6]{the second highest bid} if you win, and
\answerblank[1.4]{\(0\)} otherwise.}

\newpage
\slidehead{Should you bid what it is worth?}
\notesref{16.2.3}{Auction Games: Bayesian Nash Equilibrium for Second Pay Auction}

\vspace{0.5cm}

{\large Round 2, second price. Your value is \(v\). Let \(b\) be the highest bid
among everybody else. Compare bidding \(v\) with bidding something else.}

\vspace{0.7cm}

\begin{minipage}[t]{0.47\linewidth}
\raggedright
{\large\bfseries You bid above your value}

\vspace{0.5em}

{\large You only win extra times when \(b\) is between
\answerblank[1.4]{\(v\)} and \answerblank[1.4]{your bid}, and then you pay
\answerblank[1.4]{\(b\)} and make \answerblank[1.4]{\(v - b < 0\)}.}

\vspace{0.6em}

\bidline{3}{7}{\(b\) here: you now win, and pay more than \(v\)}
\end{minipage}
\hfill
\begin{minipage}[t]{0.47\linewidth}
\raggedright
{\large\bfseries You bid below your value}

\vspace{0.5em}

{\large You lose some auctions you would have won, and when you do still win
you pay \answerblank[3.4]{the same \(b\)}.}

\vspace{0.6em}

\bidline{7}{3}{\(b\) here: you now lose an auction you would have won at a
profit}
\end{minipage}

\vspace{0.6cm}

{\large So bidding your true value is \answerrule[5cm]{weakly dominant}.}

\newpage
\slidehead{Why first price is different}
\notesref{16.2.4}{Auction Games: Bayesian NE in a first-price auction}

\vspace{0.8cm}

{\large In round 1, bidding your value and winning scores
\answerblank[1.4]{\(0\)}, so you should bid \answerrule[4cm]{below your value}.}

\vspace{1.1cm}

{\large With \(N\) bidders and values spread evenly, the equilibrium bid is
\(b = \)\answerblank[2.6]{\(\frac{N - 1}{N}\,v\)}. For us, \(N = 3\), so
\(b = \)\answerblank[2]{\(\frac{2}{3}\,v\)}.}

\vspace{1.1cm}

{\large To bid well in a second-price auction you need to know
\answerrule[4cm]{nothing at all} about the others. In a first-price auction,
how far to shade depends on \answerrule[5cm]{how many others there are}.}

\newpage
\slidehead{What the seller made}
\notesref{16.2.5}{Auction Games: Theorem: Revenue Equivalence}

\vspace{0.6cm}

{\large The room: round 1 averaged \blank[2] and round 2 averaged \blank[2].}

\vspace{1cm}

{\large With three values spread evenly from 0 to 12, the highest is
\answerblank[1.4]{\(9\)} on average and the second highest
\answerblank[1.4]{\(6\)}.}

\vspace{0.8cm}

{\large Second price: the seller gets the second highest value,
\answerblank[1.4]{\(6\)}. \hspace{0.8cm} First price: \(\tfrac{2}{3}\) of the
highest, \answerblank[1.4]{\(6\)}.}

\vspace{1cm}

{\large In both, the item goes to the highest value and a bidder with the
lowest possible value expects nothing. So on average they raise the same:
\answerrule[5.4cm]{revenue equivalence}.}
