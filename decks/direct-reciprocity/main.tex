  \node[anchor=west, text=inkblue, font=\small\bfseries]
    at (#1,#2+0.62) {#3 v #4};
  \foreach \team [count=\r from 0] in {#3,#4}{%
    \node[anchor=east, text=inkgrey, font=\scriptsize]
      at (#1+0.95,#2-0.62*\r) {\team};
    \foreach \round in {1,...,5}{%
      \draw[inkfaint, line width=0.7pt, rounded corners=2pt, fill=white]
        (#1+0.2+0.9*\round,#2-0.62*\r-0.25)
        rectangle (#1+0.95+0.9*\round,#2-0.62*\r+0.25);}
    \draw[inkfaint, line width=0.7pt, rounded corners=2pt, fill=inksurface]
      (#1+5.85,#2-0.62*\r-0.25) rectangle (#1+7.0,#2-0.62*\r+0.25);}}
