\slidehead{Phase 1: is anybody still moving?}
\notesref{14.2.6}{Routing Games: Definition: Nash Flow}

\vspace{0.4cm}

{\large Where we settled: \answerblank[1.4]{\(1/2\)} on top,
\answerblank[1.4]{\(1/2\)} on bottom, both taking \answerblank[1.4]{\(3/2\)}
hours.}

\vspace{0.8cm}

{\large Each of us is a tiny share of the traffic: it is \textbf{non-atomic}. One
driver switching changes no road's time, so the only question is whether some
other route is already quicker.}

\vspace{0.8cm}

{\large\bfseries So this is a \answerrule[5cm]{Nash} flow.}

\vspace{0.5cm}

{\small\color{inkgrey} A flow is a Nash flow when:}\\[0.4em]
\answerspace{22.6}{2.2}{%
  Every route that carries traffic has minimal cost: if \(f_{P_1} > 0\) then
  \(c_{P_1}(f) \leq c_{P_2}(f)\), with both costs worked out at the same flow
  \(f\).}

\slidehead{Why it got worse}
\notesref{14.2.9}{Routing Games: Definition: Marginal Cost}

\vspace{0.4cm}

{\large Add a sliver \(\delta\) of traffic to a small road that already carries
\(x\).}

\vspace{0.7cm}

\hspace{1cm}
\begin{minipage}{0.9\linewidth}
  {\large The sliver itself spends \answerblank[1.4]{\(x\)} hours per driver:
   \answerblank[1.4]{\(x\delta\)} in all.}\\[0.9em]
  {\large The \(x\) already there are each slowed by \answerblank[1.4]{\(\delta\)}:
   \answerblank[1.4]{\(x\delta\)} in all.}
\end{minipage}

\vspace{0.7cm}

{\large A driver feels the first and ignores the second. The road's total time is}

\vspace{0.5cm}

\begin{center}
{\Large $x \, c(x) \;=\;$ \answerblank[2]{\(x^{2}\)} \hspace{2.2cm}
 and the sliver really adds
 $\dfrac{\mathrm{d}}{\mathrm{d}x}\big(x \, c(x)\big)\,\delta \;=\;$
 \answerblank[2]{\(2x\)}$\,\delta$}
\end{center}

\vspace{0.8cm}

{\large With everyone on the small roads, \(x = 1\): each driver feels
\answerblank[1.4]{\(1\)} hour but costs everyone \answerblank[1.4]{\(2\)}.}

\slidehead{Two theorems}
\notesref{14.2.7 to 14.2.10}{Routing Games: Potential Function, Marginal Cost}

\vspace{0.2cm}

\begin{minipage}[t]{0.48\linewidth}
\tikz\node[draw=inkteal, line width=1pt, rounded corners=4pt, fill=white,
  text width=10.7cm, inner sep=0.45cm, align=left, text=inktext]{%
  {\bfseries\color{inkteal} Theorem.} A Nash flow minimises the potential
  \[
    \Phi(f) = \sum_{e} \int_{0}^{f_e} c_e(x)\,\mathrm{d}x.
  \]};

\vspace{0.4cm}

{\large Everyone on the new route: \(\Phi =\) \answerblank[1.6]{\(1\)}}

\vspace{0.6em}

{\large Half each way: \(\Phi =\) \answerblank[1.6]{\(5/4\)}}

\vspace{0.6em}

{\large\color{inkgrey} The room minimised \(\Phi\), not the time. An
integral, not a sum: the traffic is non-atomic.}
\end{minipage}
\hfill
\begin{minipage}[t]{0.48\linewidth}
\tikz\node[draw=inkred, line width=1pt, rounded corners=4pt, fill=white,
  text width=10.7cm, inner sep=0.45cm, align=left, text=inktext]{%
  {\bfseries\color{inkred} Theorem.} The optimal flow is the Nash flow once
  every cost is replaced by its marginal cost
  \[
    c^{*}_e(x) = \frac{\mathrm{d}}{\mathrm{d}x}\big(x\,c_e(x)\big).
  \]};

\vspace{0.1cm}

\begin{center}

  \newroad{\answerblank[1.2]{\(0\)}}}
\end{center}
\end{minipage}

{\large At \((\tfrac12, \tfrac12, 0)\) every route costs
\answerblank[1.4]{\(2\)} in marginal hours, so nobody gains by moving.}
